4. Single-File JavaScript Engine Prototype
This lightweight JavaScript module handles barrel length FPS scaling, COAL reduction pressure warnings, and ladder generation.

JavaScript
/**
 * Apex AMC - Internal Ballistics & Ladder Calculation Engine
 */

// Sample Published Recipe
const sampleRecipe = {
  cartridge: ".308 Winchester",
  bullet: "168gr HPBT",
  powder: "I4064",
  testBarrelLength: 24, // inches
  publishedCOAL: 2.800, // inches
  minChargeGrains: 41.5,
  minFPS: 2450,
  maxChargeGrains: 45.0,
  maxFPS: 2700,
  maxPSI: 60000,
  burnRateClass: "rifle_medium" // exponent n ~ 0.22
};

function calculateReloadingLadder(recipe, userInputs) {
  const { userBarrelLength, userTargetCOAL, stepsCount = 5 } = userInputs;

  // 1. Exponent Selection for Barrel Scaling
  const exponentMap = {
    pistol: 0.15,
    rifle_medium: 0.22,
    rifle_slow: 0.29
  };
  const n = exponentMap[recipe.burnRateClass] || 0.22;

  // 2. Barrel Length Scaling Factor
  const barrelRatio = userBarrelLength / recipe.testBarrelLength;
  const fpsScaleFactor = Math.pow(barrelRatio, n);

  // 3. COAL Pressure Warning Check
  let coalWarning = null;
  if (userTargetCOAL < recipe.publishedCOAL) {
    const diff = (recipe.publishedCOAL - userTargetCOAL).toFixed(3);
    coalWarning = `WARNING: Target COAL is ${diff}" shorter than published COAL (${recipe.publishedCOAL}"). Seating deeper decreases internal volume and increases peak chamber pressure sharply! Start below minimum charge.`;
  }

  // 4. Generate Ladder Steps
  const stepIncrement = (recipe.maxChargeGrains - recipe.minChargeGrains) / (stepsCount - 1);
  const fpsIncrement = (recipe.maxFPS - recipe.minFPS) / (stepsCount - 1);

  const ladder = [];
  for (let i = 0; i < stepsCount; i++) {
    const currentCharge = recipe.minChargeGrains + (stepIncrement * i);
    const baseFPS = recipe.minFPS + (fpsIncrement * i);
    const estimatedFPS = Math.round(baseFPS * fpsScaleFactor);

    ladder.push({
      step: i + 1,
      chargeGrains: Number(currentCharge.toFixed(1)),
      estimatedFPS: estimatedFPS,
      isMax: i === stepsCount - 1
    });
  }

  return {
    cartridge: recipe.cartridge,
    bullet: recipe.bullet,
    powder: recipe.powder,
    publishedCOAL: recipe.publishedCOAL,
    targetCOAL: userTargetCOAL,
    testBarrelLength: recipe.testBarrelLength,
    userBarrelLength: userBarrelLength,
    coalWarning: coalWarning,
    ladder: ladder
  };
}

// Example Run:
const result = calculateReloadingLadder(sampleRecipe, {
  userBarrelLength: 18, // 18-inch barrel instead of 24-inch
  userTargetCOAL: 2.780, // Slightly shorter seating depth
  stepsCount: 5
});

How do I design a relational SQL schema and JSON importer for published reloading recipes?

Designing a relational database for reloading recipes requires a normalized schema that prevents data duplication while allowing instant lookups across calibers, bullets, powders, and manufacturers.

Here is a complete PostgreSQL/SQLite-compatible relational schema, a sample JSON data payload format, and a zero-dependency Python importer script.

1. Relational SQL Schema (schema.sql)
This schema splits physical components (bullets, powders, cartridges) into lookups to ensure that querying by a specific bullet or powder searches across all available recipes.

SQL
-- Cartridges / Calibers (e.g., .308 Winchester, 9mm Luger)
CREATE TABLE IF NOT EXISTS cartridges (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(100) NOT NULL UNIQUE,
    category VARCHAR(50) DEFAULT 'rifle' CHECK(category IN ('rifle', 'handgun', 'shotgun')),
    saami_max_psi INTEGER
);

-- Component Manufacturers (e.g., Sierra, Hodgdon, Hornady, Federal)
CREATE TABLE IF NOT EXISTS manufacturers (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(100) NOT NULL UNIQUE
);

-- Bullets / Projectiles
CREATE TABLE IF NOT EXISTS bullets (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    manufacturer_id INTEGER NOT NULL,
    name VARCHAR(100) NOT NULL,
    weight_grains DECIMAL(6,2) NOT NULL,
    diameter_inches DECIMAL(5,3) NOT NULL,
    bc_g1 DECIMAL(5,3),
    bc_g7 DECIMAL(5,3),
    FOREIGN KEY (manufacturer_id) REFERENCES manufacturers(id),
    UNIQUE(manufacturer_id, name, weight_grains)
);

-- Powders
CREATE TABLE IF NOT EXISTS powders (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    manufacturer_id INTEGER NOT NULL,
    name VARCHAR(100) NOT NULL,
    burn_rate_class VARCHAR(50) CHECK(burn_rate_class IN ('pistol', 'rifle_medium', 'rifle_slow', 'shotgun')),
    FOREIGN KEY (manufacturer_id) REFERENCES manufacturers(id),
    UNIQUE(manufacturer_id, name)
);

-- Data Sources / Manuals (Attribution & Compliance)
CREATE TABLE IF NOT EXISTS sources (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(150) NOT NULL,
    edition VARCHAR(50),
    publication_year INTEGER,
    url TEXT,
    UNIQUE(name, edition)
);

-- Core Reloading Recipes Table
CREATE TABLE IF NOT EXISTS recipes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    cartridge_id INTEGER NOT NULL,
    bullet_id INTEGER NOT NULL,
    powder_id INTEGER NOT NULL,
    source_id INTEGER NOT NULL,
    primer_description VARCHAR(100),
    case_brand VARCHAR(100),
    test_barrel_length_in DECIMAL(4,2) NOT NULL,
    test_twist_rate VARCHAR(20),
    published_coal_in DECIMAL(5,3) NOT NULL,
    min_charge_grains DECIMAL(5,2) NOT NULL,
    min_velocity_fps INTEGER NOT NULL,
    max_charge_grains DECIMAL(5,2) NOT NULL,
    max_velocity_fps INTEGER NOT NULL,
    max_pressure INTEGER,
    pressure_unit VARCHAR(10) DEFAULT 'PSI' CHECK(pressure_unit IN ('PSI', 'CUP')),
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

This Python script reads recipes.json, initializes an SQLite database (reloading.db), handles normalized "get-or-create" logic for foreign keys, and bulk-inserts recipes safely.

Python
import json
import sqlite3
from typing import Any, Dict

DB_NAME = "reloading.db"
JSON_FILE = "recipes.json"

def get_or_create(cursor: sqlite3.Cursor, table: str, search_fields: Dict[str, Any], insert_fields: Dict[str, Any]) -> int:
    """Helper to select an existing ID or insert a new row and return the ID."""
    where_clause = " AND ".join([f"{k} = ?" for k in search_fields.keys()])
    query = f"SELECT id FROM {table} WHERE {where_clause}"
    cursor.execute(query, list(search_fields.values()))
    row = cursor.fetchone()
    
    if row:
        return row[0]
    
    keys = ", ".join(insert_fields.keys())
    placeholders = ", ".join(["?"] * len(insert_fields))
    insert_query = f"INSERT INTO {table} ({keys}) VALUES ({placeholders})"
    cursor.execute(insert_query, list(insert_fields.values()))
    return cursor.lastrowid

def import_json_data():
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    
    # 1. Initialize schema
    with open("schema.sql", "r") as f:
        cursor.executescript(f.read())
        
    # 2. Load JSON recipes
    with open(JSON_FILE, "r") as f:
        recipes_data = json.load(f)
        
    count = 0
    for r in recipes_data:
        # Resolve Cartridge
        cartridge_id = get_or_create(
            cursor, "cartridges",
            {"name": r["cartridge"]},
            {"name": r["cartridge"], "category": r.get("cartridge_category", "rifle"), "saami_max_psi": r.get("saami_max_psi")}
        )
        
        # Resolve Source
        source_id = get_or_create(
            cursor, "sources",
            {"name": r["source"]["name"], "edition": r["source"].get("edition", "Default")},
            {"name": r["source"]["name"], "edition": r["source"].get("edition", "Default"), "publication_year": r["source"].get("publication_year")}
        )
        
        # Resolve Bullet Manufacturer
        bullet_mfg_id = get_or_create(
            cursor, "manufacturers",
            {"name": r["bullet"]["manufacturer"]},
            {"name": r["bullet"]["manufacturer"]}
        )
        
        # Resolve Bullet
        bullet_id = get_or_create(
            cursor, "bullets",
            {"manufacturer_id": bullet_mfg_id, "name": r["bullet"]["name"], "weight_grains": r["bullet"]["weight_grains"]},
            {
                "manufacturer_id": bullet_mfg_id,
                "name": r["bullet"]["name"],
                "weight_grains": r["bullet"]["weight_grains"],
                "diameter_inches": r["bullet"]["diameter_inches"],
                "bc_g1": r["bullet"].get("bc_g1"),
                "bc_g7": r["bullet"].get("bc_g7")
            }
        )
        
        # Resolve Powder Manufacturer
        powder_mfg_id = get_or_create(
            cursor, "manufacturers",
            {"name": r["powder"]["manufacturer"]},
            {"name": r["powder"]["manufacturer"]}
        )
        
        # Resolve Powder
        powder_id = get_or_create(
            cursor, "powders",
            {"manufacturer_id": powder_mfg_id, "name": r["powder"]["name"]},
            {
                "manufacturer_id": powder_mfg_id,
                "name": r["powder"]["name"],
                "burn_rate_class": r["powder"].get("burn_rate_class", "rifle_medium")
            }
        )
        
        # Insert Recipe
        cursor.execute("""
            INSERT INTO recipes (
                cartridge_id, bullet_id, powder_id, source_id, primer_description,
                case_brand, test_barrel_length_in, test_twist_rate, published_coal_in,
                min_charge_grains, min_velocity_fps, max_charge_grains, max_velocity_fps,
                max_pressure, pressure_unit, notes
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            cartridge_id, bullet_id, powder_id, source_id, r.get("primer"),
            r.get("case_brand"), r["test_barrel_length_in"], r.get("test_twist_rate"),
            r["published_coal_in"], r["min_charge_grains"], r["min_velocity_fps"],
            r["max_charge_grains"], r["max_velocity_fps"], r.get("max_pressure"),
            r.get("pressure_unit", "PSI"), r.get("notes")
        ))
        count += 1

    conn.commit()
    conn.close()
    print(f"Successfully imported {count} recipes into {DB_NAME}.")

How do I build a FastAPI endpoint that queries this SQLite database and returns scaled ladder data for a given user barrel length?

Here is a complete FastAPI implementation in Python that connects to the reloading.db SQLite database, queries a published recipe, applies the velocity scaling formula based on the user's custom barrel length, checks for seating depth safety warnings, and returns a structured ladder response.

1. Requirements
Install fastapi and uvicorn:

2. FastAPI Application (main.py)
Python
import math
import sqlite3
from typing import List, Optional
from fastapi import FastAPI, HTTPException, Query
from pydantic import BaseModel, Field

DB_NAME = "reloading.db"

app = FastAPI(
    title="Apex AMC - Reloading Recipe & Ladder API",
    description="Query published load data and calculate barrel-length scaled work-up ladders.",
    version="1.0.0"
)

# --- Pydantic Response & Request Schemas ---

class LadderStep(BaseModel):
    step: int
    charge_grains: float = Field(..., description="Powder charge weight in grains")
    estimated_fps: int = Field(..., description="Estimated muzzle velocity in feet per second")
    is_max: bool = Field(..., description="Flag indicating if this is the maximum safe published load")

class LadderResponse(BaseModel):
    recipe_id: int
    cartridge: str
    bullet: str
    powder: str
    source: str
    published_coal_in: float
    user_target_coal_in: float
    test_barrel_length_in: float
    user_barrel_length_in: float
    coal_warning: Optional[str] = None
    ladder: List[LadderStep]


# --- Helper Functions ---

def get_db_connection():
    conn = sqlite3.connect(DB_NAME)
    conn.row_factory = sqlite3.Row  # Enables dict-like column access
    return conn


def calculate_velocity_scaling(published_fps: float, test_length: float, user_length: float, burn_class: str) -> float:
    """Homann / Fraser Expansion Model: V_target = V_test * (L_target / L_test)^n"""
    exponent_map = {
        "pistol": 0.15,
        "rifle_medium": 0.22,
        "rifle_slow": 0.29,
        "shotgun": 0.15
    }
    n = exponent_map.get(burn_class, 0.22)
    scaling_factor = math.pow(user_length / test_length, n)
    return published_fps * scaling_factor


# --- API Endpoints ---

@app.get("/recipes/{recipe_id}/ladder", response_model=LadderResponse, summary="Get Scaled Reloading Ladder")
def get_recipe_ladder(
    recipe_id: int,
    user_barrel_length: float = Query(..., gt=0, description="User's firearm barrel length in inches (e.g., 18.0)"),
    user_target_coal: Optional[float] = Query(None, gt=0, description="User's intended Overall Length in inches (e.g., 2.780)"),
    steps_count: int = Query(5, ge=3, le=10, description="Number of ladder work-up steps (between 3 and 10)")
):
    conn = get_db_connection()
    cursor = conn.cursor()

    # Query normalized tables via joins
    query = """
        SELECT 
            r.id as recipe_id,
            r.published_coal_in,
            r.test_barrel_length_in,
            r.min_charge_grains,
            r.min_velocity_fps,
            r.max_charge_grains,
            r.max_velocity_fps,
            c.name as cartridge_name,
            b.weight_grains as bullet_weight,
            b.name as bullet_name,
            bm.name as bullet_mfg,
            p.name as powder_name,
            p.burn_rate_class,
            pm.name as powder_mfg,
            s.name as source_name,
            s.edition as source_edition
        FROM recipes r
        JOIN cartridges c ON r.cartridge_id = c.id
        JOIN bullets b ON r.bullet_id = b.id
        JOIN manufacturers bm ON b.manufacturer_id = bm.id
        JOIN powders p ON r.powder_id = p.id
        JOIN manufacturers pm ON p.manufacturer_id = pm.id
        JOIN sources s ON r.source_id = s.id
        WHERE r.id = ?
    """
    cursor.execute(query, (recipe_id,))
    row = cursor.fetchone()
    conn.close()

    if not row:
        raise HTTPException(status_code=404, detail=f"Recipe ID {recipe_id} not found.")

    published_coal = row["published_coal_in"]
    target_coal = user_target_coal if user_target_coal is not None else published_coal

    # COAL Seating Depth Safety Warning
    coal_warning = None
    if target_coal < published_coal:
        diff = round(published_coal - target_coal, 3)
        coal_warning = (
            f"WARNING: Target COAL ({target_coal}\") is {diff}\" shorter than published COAL ({published_coal}\"). "
            "Seating deeper reduces internal combustion volume and rapidly spikes chamber pressure! "
            "Always begin testing at or below minimum charge."
        )

    # Calculate Velocity Scaling Ratios
    min_scaled_fps = calculate_velocity_scaling(
        row["min_velocity_fps"], row["test_barrel_length_in"], user_barrel_length, row["burn_rate_class"]
    )
    max_scaled_fps = calculate_velocity_scaling(
        row["max_velocity_fps"], row["test_barrel_length_in"], user_barrel_length, row["burn_rate_class"]
    )

    # Compute Linear Ladder Increments
    charge_step = (row["max_charge_grains"] - row["min_charge_grains"]) / (steps_count - 1)
    fps_step = (max_scaled_fps - min_scaled_fps) / (steps_count - 1)

    ladder_steps = []
    for i in range(steps_count):
        current_charge = row["min_charge_grains"] + (charge_step * i)
        current_fps = min_scaled_fps + (fps_step * i)
        
        ladder_steps.append(
            LadderStep(
                step=i + 1,
                charge_grains=round(current_charge, 1),
                estimated_fps=round(current_fps),
                is_max=(i == steps_count - 1)
            )
        )

    return LadderResponse(
        recipe_id=row["recipe_id"],
        cartridge=row["cartridge_name"],
        bullet=f"{row['bullet_mfg']} {row['bullet_name']} ({row['bullet_weight']}gr)",
        powder=f"{row['powder_mfg']} {row['powder_name']}",
        source=f"{row['source_name']} ({row['source_edition']})",
        published_coal_in=published_coal,
        user_target_coal_in=target_coal,
        test_barrel_length_in=row["test_barrel_length_in"],
        user_barrel_length_in=user_barrel_length,
        coal_warning=coal_warning,
        ladder=ladder_steps
    )

SCHEMA_SQL = """
-- Cartridges / Calibers

    model_name VARCHAR(100) NOT NULL,
    primer_type VARCHAR(50) NOT NULL,

);

-- Powders
CREATE TABLE IF NOT EXISTS powders (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    manufacturer_id INTEGER NOT NULL,
    name VARCHAR(100) NOT NULL,
    burn_rate_class VARCHAR(50) CHECK(burn_rate_class IN ('pistol', 'rifle_medium', 'rifle_slow', 'shotgun')),
    FOREIGN KEY (manufacturer_id) REFERENCES manufacturers(id),
    UNIQUE(manufacturer_id, name)
);

-- Published Sources / Manuals
CREATE TABLE IF NOT EXISTS sources (

    name VARCHAR(150) NOT NULL,
    edition VARCHAR(50),
    publication_year INTEGER,
    UNIQUE(name, edition)
);

-- Master Recipes Table
CREATE TABLE IF NOT EXISTS recipes (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    cartridge_id INTEGER NOT NULL,
    bullet_id INTEGER NOT NULL,
    powder_id INTEGER NOT NULL,
    primer_id INTEGER NOT NULL,

);

CREATE INDEX IF NOT EXISTS idx_bullets_search ON bullets(manufacturer_id, weight_grains);
CREATE INDEX IF NOT EXISTS idx_powders_search ON powders(manufacturer_id);
"""

def get_or_create(cursor, table, search_fields, insert_fields):
    where_clause = " AND ".join([f"{k} = ?" for k in search_fields.keys()])
    query = f"SELECT id FROM {table} WHERE {where_clause}"
    cursor.execute(query, list(search_fields.values()))
    row = cursor.fetchone()
    
    if row:
        return row[0]
    
    keys = ", ".join(insert_fields.keys())
    placeholders = ", ".join(["?"] * len(insert_fields))
    insert_query = f"INSERT INTO {table} ({keys}) VALUES ({placeholders})"
    cursor.execute(insert_query, list(insert_fields.values()))
    return cursor.lastrowid

def build_database():

import sqlite3

DB_NAME = "master_reloading.db"

def search_recipes(cartridge_query=None, powder_query=None):
    conn = sqlite3.connect(DB_NAME)
    conn.row_factory = sqlite3.Row
    cursor = conn.cursor()

    sql = """
        SELECT 
            c.name as cartridge,
            bm.name || ' ' || b.name || ' (' || b.weight_grains || 'gr)' as bullet,
            pm.name || ' ' || p.name as powder,
            prm.name || ' ' || pr.model_name as primer,
            r.min_charge_grains || ' - ' || r.max_charge_grains || ' gr' as charge_range,
            r.min_velocity_fps || ' - ' || r.max_velocity_fps || ' fps' as velocity_range,
            s.name || ' (' || s.edition || ')' as source
        FROM recipes r
        JOIN cartridges c ON r.cartridge_id = c.id
        JOIN bullets b ON r.bullet_id = b.id
        JOIN manufacturers bm ON b.manufacturer_id = bm.id
        JOIN powders p ON r.powder_id = p.id
        JOIN manufacturers pm ON p.manufacturer_id = pm.id
        JOIN primers pr ON r.primer_id = pr.id
        JOIN manufacturers prm ON pr.manufacturer_id = prm.id
        JOIN sources s ON r.source_id = s.id
        WHERE 1=1
    """
    params = []
    if cartridge_query:
        sql += " AND c.name LIKE ?"
        params.append(f"%{cartridge_query}%")
    if powder_query:
        sql += " AND p.name LIKE ?"
        params.append(f"%{powder_query}%")

    cursor.execute(sql, params)
    rows = cursor.fetchall()
    conn.close()

    print(f"\n--- FOUND {len(rows)} MATCHING MASTER LOAD RECIPES ---")
    for r in rows:
        print(f"\n[{r['cartridge']}]")
        print(f"  Bullet:   {r['bullet']}")
        print(f"  Powder:   {r['powder']}")
        print(f"  Primer:   {r['primer']}")
        print(f"  Charge:   {r['charge_range']}")
        print(f"  Velocity: {r['velocity_range']}")
        print(f"  Source:   {r['source']}")

if __name__ == "__main__":
    # Example Searches:
    search_recipes(cartridge_query="308")
    search_recipes(powder_query="Titegroup")

// --- Statistical Helper Functions ---
const calculateStats = (shots) => {
  const valid = shots.map(Number).filter(n => !isNaN(n) && n > 0);
  if (valid.length === 0) return { avg: 0, es: 0, sd: 0, count: 0 };
  
  const count = valid.length;
  const sum = valid.reduce((a, b) => a + b, 0);
  const avg = sum / count;
  const max = Math.max(...valid);
  const min = Math.min(...valid);
  const es = max - min;
  
  let sd = 0;
  if (count > 1) {
    const variance = valid.reduce((a, b) => a + Math.pow(b - avg, 2), 0) / (count - 1);
    sd = Math.sqrt(variance);
  }
  
  return { 
    avg: Math.round(avg), 
    es: Math.round(es), 
    sd: Math.round(sd * 10) / 10, 
    count 
  };
};

    if (!activeRecipe) return [];
    
    const userBarrel = parseFloat(activeProfile.barrelLength) || activeRecipe.testBarrel;
    const userCoal = parseFloat(activeProfile.targetCoal) || activeRecipe.publishedCOAL;

    const expMap = { pistol: 0.15, rifle_medium: 0.22, rifle_slow: 0.29 };
    const n = expMap[activeRecipe.burnRateClass] || 0.22;
    const scaleFactor = Math.pow(userBarrel / activeRecipe.testBarrel, n);

    const minScaledFPS = activeRecipe.minFPS * scaleFactor;
    const maxScaledFPS = activeRecipe.maxFPS * scaleFactor;

    const stepsCount = 5;
    const chargeInc = (activeRecipe.maxCharge - activeRecipe.minCharge) / (stepsCount - 1);
    const fpsInc = (maxScaledFPS - minScaledFPS) / (stepsCount - 1);

    const ladder = [];
    for (let i = 0; i < stepsCount; i++) {
      const charge = activeRecipe.minCharge + (chargeInc * i);
      const estFPS = minScaledFPS + (fpsInc * i);
      ladder.push({
        step: i + 1,
        charge: charge.toFixed(1),
        estFPS: Math.round(estFPS),
        isMax: i === stepsCount - 1
      });

    return ladder;

New Capabilities Added
Load Profile Manager: Shooters can create distinct profiles (e.g., "Tikka T3x .308 Match", "AR-15 11.5 SBR") mapping specific firearms, custom barrel lengths, and seating depths (COAL) to published recipes.

Shot-by-Shot Chrono Logging: Allows typing velocity readings for individual shots (e.g., Shot 1 = 2452, Shot 2 = 2458, Shot 3 = 2449). Tap any shot tag to delete an anomaly.

Real-time Statistical Computations:

AVG (Average Velocity): Sum divided by total valid shots.

ES (Extreme Spread): Maximum Velocity−Minimum Velocity.

SD (Standard Deviation): Sample standard deviation calculated on-the-fly to indicate velocity consistency.

// --- Simplified Ballistic Point-Mass Drop & Windage Solver ---
// Calculates Drop (in inches/MIL/MOA) and Windage Drift
const calculateDope = (distanceYards, velocityFps, bcG1, sightHeightInches, windSpeedMph, windAngleDeg, tempF, altitudeFt) => {
  if (!distanceYards || distanceYards <= 0) return { dropInches: 0, milDrop: 0, moaDrop: 0, milWind: 0, moaWind: 0 };

  // Environmental density correction factor (Simplified DA adjustment)
  const tempCorrection = 1 + ((tempF - 59) * 0.002);
  const altCorrection = 1 - (altitudeFt * 0.00002);
  const effectiveBC = bcG1 * tempCorrection * altCorrection;

  // Approximate time of flight (t = d / v_avg)
  const distanceFeet = distanceYards * 3;
  // Drag decay factor
  const k = 0.00008 / Math.max(effectiveBC, 0.1);
  const estTimeOfFlight = (Math.exp(k * distanceFeet) - 1) / (k * velocityFps);

  // Gravity Drop calculation: h = 0.5 * g * t^2
  const gravityInches = 0.5 * 386.1 * Math.pow(estTimeOfFlight, 2);
  
  // Sight line intersection compensation (100 yard zero assumption)
  const zeroTime = (300) / velocityFps;
  const zeroGravity = 0.5 * 386.1 * Math.pow(zeroTime, 2);
  const netDropInches = gravityInches - ((distanceYards / 100) * (zeroGravity + sightHeightInches)) + sightHeightInches;

  // Crosswind drift calculation (Crosswind component = sin(angle))
  const crossWindMph = windSpeedMph * Math.sin((windAngleDeg * Math.PI) / 180);
  const crossWindFps = crossWindMph * 1.46667;
  // Lag law for wind drift: Drift = w * (t - d/v0)
  const windDriftInches = Math.abs(crossWindFps * (estTimeOfFlight - (distanceFeet / velocityFps)) * 12);

  // Convert Inches to MIL and MOA
  // 1 MIL @ 100 yards = 3.6 inches. 1 MOA @ 100 yards = 1.047 inches.
  const yardFactor = distanceYards / 100;
  const milDrop = netDropInches / (3.6 * yardFactor);
  const moaDrop = netDropInches / (1.047 * yardFactor);
  
  const milWind = windDriftInches / (3.6 * yardFactor);
  const moaWind = windDriftInches / (1.047 * yardFactor);

  return {
    dropInches: netDropInches.toFixed(1),
    milDrop: Math.max(0, milDrop).toFixed(1),
    moaDrop: Math.max(0, moaDrop).toFixed(1),
    milWind: milWind.toFixed(1),
    moaWind: moaWind.toFixed(1),
    timeOfFlight: estTimeOfFlight.toFixed(2)
  };
};

React Native Target Group Calculator Component
This component uses Pressable to register tap relative coordinates, toggles between reference scaling and shot placement, and dynamically renders marker overlays and bounding lines.

JavaScript
import React, { useState } from 'react';
import {
  StyleSheet,
  Text,
  View,
  Image,
  Pressable,
  TextInput,
  TouchableOpacity,
  SafeAreaView,
  ScrollView
} from 'react-native';

export default function TargetGroupCalculator() {
  // Input parameters
  const [targetDistanceYards, setTargetDistanceYards] = useState('100');
  const [refSizeInches, setRefSizeInches] = useState('1.0');
  
  // Interaction mode: 'SCALE' (tap 2 points) or 'SHOTS' (tap bullet holes)
  const [mode, setMode] = useState('SCALE'); 

  // Tap coordinates
  const [scalePoints, setScalePoints] = useState([]); // [{x, y}, {x, y}]
  const [shots, setShots] = useState([]);             // [{x, y}, ...]

  // Local sample target image URL
  const [imageUri] = useState('https://images.unsplash.com/photo-1595590424283-b8f17842773f?auto=format&fit=crop&w=800&q=80');

  // Tap handler on the target image
  const handleImagePress = (event) => {
    const { locationX, locationY } = event.nativeEvent;
    const newPoint = { x: locationX, y: locationY };

    if (mode === 'SCALE') {
      if (scalePoints.length < 2) {
        setScalePoints([...scalePoints, newPoint]);
      } else {
        // Reset scale if tapped again
        setScalePoints([newPoint]);
      }
    } else if (mode === 'SHOTS') {
      setShots([...shots, newPoint]);
    }
  };

  // Geometry calculations
  const calculateResults = () => {
    if (scalePoints.length < 2) {
      return { ppi: 0, groupInches: 0, moa: 0, mil: 0, maxPair: null };
    }

    // 1. Calculate Pixels Per Inch (PPI)
    const dxScale = scalePoints[1].x - scalePoints[0].x;
    const dyScale = scalePoints[1].y - scalePoints[0].y;
    const pixelDistanceRef = Math.sqrt(dxScale * dxScale + dyScale * dyScale);
    const refInches = parseFloat(refSizeInches) || 1.0;
    const ppi = pixelDistanceRef / refInches;

    if (shots.length < 2 || ppi === 0) {
      return { ppi, groupInches: 0, moa: 0, mil: 0, maxPair: null };
    }

    // 2. Find Maximum Extreme Spread among shots
    let maxPxDist = 0;
    let maxPair = null;

    for (let i = 0; i < shots.length; i++) {
      for (let j = i + 1; j < shots.length; j++) {
        const dx = shots[j].x - shots[i].x;
        const dy = shots[j].y - shots[i].y;
        const dist = Math.sqrt(dx * dx + dy * dy);
        if (dist > maxPxDist) {
          maxPxDist = dist;
          maxPair = [shots[i], shots[j]];
        }
      }
    }

    // 3. Convert to Inches, MOA, and MILs
    const groupInches = maxPxDist / ppi;
    const yards = parseFloat(targetDistanceYards) || 100;
    const moa = groupInches / (yards * 0.010472);
    const mil = groupInches / (yards * 0.036);

    return {
      ppi: ppi.toFixed(1),
      groupInches: groupInches.toFixed(2),
      moa: moa.toFixed(2),
      mil: mil.toFixed(2),
      maxPair
    };
  };

  const results = calculateResults();

  const resetAll = () => {
    setScalePoints([]);
    setShots([]);
    setMode('SCALE');
  };

  return (
    <SafeAreaView style={styles.container}>
      <ScrollView contentContainerStyle={styles.content}>
        
        <Text style={styles.title}>Target Group Analyzer</Text>

        {/* Target Image Touch Canvas */}
        <View style={styles.imageContainer}>
          <Pressable onPress={handleImagePress} style={styles.pressableArea}>
            <Image
              source={{ uri: imageUri }}
              style={styles.targetImage}
              resizeMode="contain"
            />

            {/* Render Calibration Markers (Blue) */}
            {scalePoints.map((pt, idx) => (
              <View
                key={`scale-${idx}`}
                style={[styles.scaleMarker, { left: pt.x - 8, top: pt.y - 8 }]}
              >
                <Text style={styles.markerLabel}>R{idx + 1}</Text>
              </View>
            ))}

            {/* Render Shot Markers (Red) */}
            {shots.map((pt, idx) => (
              <View
                key={`shot-${idx}`}
                style={[styles.shotMarker, { left: pt.x - 10, top: pt.y - 10 }]}
              >
                <Text style={styles.shotLabel}>{idx + 1}</Text>
              </View>
            ))}
          </Pressable>
        </View>

        {/* Mode Controls */}
        <View style={styles.modeRow}>
          <TouchableOpacity
            style={[styles.modeBtn, mode === 'SCALE' && styles.activeModeBtn]}
            onPress={() => setMode('SCALE')}
          >
            <Text style={styles.modeBtnText}>
              1. CALIBRATE SCALE ({scalePoints.length}/2)
            </Text>
          </TouchableOpacity>

          <TouchableOpacity
            style={[styles.modeBtn, mode === 'SHOTS' && styles.activeModeBtn]}
            onPress={() => setMode('SHOTS')}
          >
            <Text style={styles.modeBtnText}>
              2. TAP SHOTS ({shots.length})
            </Text>
          </TouchableOpacity>
        </View>

        {/* Parameters & Inputs */}
        <View style={styles.inputCard}>
          <View style={styles.inputRow}>
            <View style={styles.inputGroup}>
              <Text style={styles.label}>Known Ref Size (in)</Text>
              <TextInput
                style={styles.input}
                keyboardType="numeric"
                value={refSizeInches}
                onChangeText={setRefSizeInches}
              />
            </View>
            <View style={styles.inputGroup}>
              <Text style={styles.label}>Target Distance (yds)</Text>
              <TextInput
                style={styles.input}
                keyboardType="numeric"
                value={targetDistanceYards}
                onChangeText={setTargetDistanceYards}

JavaScript Calculation Engine (ballisticsStats.js)
Below is the standalone calculation utility:

JavaScript
/**
 * Calculates POI, Mean Radius, and Turret Click Corrections
 */
export const calculateAdvancedStats = ({
  shots,            // Array of [{x, y}]
  poa,              // Point of Aim {x, y}
  ppi,              // Pixels Per Inch
  distanceYards,    // Distance in yards
  unit = 'MOA',     // 'MOA' | 'MIL'
  clickValue = 0.25 // 0.25 for 1/4 MOA, 0.1 for 0.1 MIL
}) => {
  if (!shots || shots.length === 0 || !ppi || ppi <= 0) {
    return null;
  }

  const n = shots.length;

  // 1. Calculate Center of Impact (POI)
  const poi = shots.reduce(
    (acc, shot) => ({ x: acc.x + shot.x / n, y: acc.y + shot.y / n }),
    { x: 0, y: 0 }
  );

  // 2. Calculate Mean Radius (MR)
  const sumRadiusPx = shots.reduce((sum, shot) => {
    const dx = shot.x - poi.x;
    const dy = shot.y - poi.y;
    return sum + Math.sqrt(dx * dx + dy * dy);
  }, 0);

  const meanRadiusPx = sumRadiusPx / n;
  const meanRadiusInches = meanRadiusPx / ppi;

  // Angular factor per 100 yards
  const angularFactor = unit === 'MOA' ? 0.010472 : 0.036;
  const inchesPerAngularUnit = (distanceYards / 100) * (angularFactor * 100);
  const meanRadiusAngular = meanRadiusInches / (distanceYards * angularFactor);

  // 3. Zero Offset (POI relative to POA)
  let zeroOffset = null;
  if (poa) {
    const dxPx = poi.x - poa.x;      // Positive = Right
    const dyPx = poa.y - poi.y;      // Positive = High (Screen y inverted)

    const dxInches = dxPx / ppi;
    const dyInches = dyPx / ppi;

    const dxAngular = dxInches / (distanceYards * angularFactor);
    const dyAngular = dyInches / (distanceYards * angularFactor);

    // Clicks calculation
    const hClicks = Math.round(Math.abs(dxAngular) / clickValue);
    const vClicks = Math.round(Math.abs(dyAngular) / clickValue);

    zeroOffset = {
      dxInches: dxInches.toFixed(2),
      dyInches: dyInches.toFixed(2),
      dxAngular: Math.abs(dxAngular).toFixed(2),
      dyAngular: Math.abs(dyAngular).toFixed(2),
      // Turret Instructions: Correct opposite to impact
      hDirection: dxInches > 0 ? 'LEFT' : dxInches < 0 ? 'RIGHT' : 'ZEROED',
      vDirection: dyInches > 0 ? 'DOWN' : dyInches < 0 ? 'UP' : 'ZEROED',
      hClicks,
      vClicks,
    };
  }

  return {
    poi,
    meanRadiusInches: meanRadiusInches.toFixed(2),
    meanRadiusAngular: meanRadiusAngular.toFixed(2),
    zeroOffset,
  };
};

Code Implementation: Outlier/Flyer Detection (Grubbs' Test)
Here is a ready-to-use utility module for detecting outliers in bullet group data:

JavaScript
/**
 * Detects statistical flyers using Grubbs' Test for Outliers
 * @param {Array<{x: number, y: number}>} shots - Array of shot coordinates
 * @param {number} ppi - Pixels per inch
 * @param {number} alpha - Significance level (default 0.05)
 */
export const detectFlyers = (shots, ppi, alpha = 0.05) => {
  if (!shots || shots.length < 3) return { cleanShots: shots, flyers: [] };

  const n = shots.length;

  // 1. Calculate POI
  const poi = shots.reduce(
    (acc, s) => ({ x: acc.x + s.x / n, y: acc.y + s.y / n }),
    { x: 0, y: 0 }
  );

  // 2. Calculate distance of each shot from POI
  const radii = shots.map((s, index) => {
    const dx = s.x - poi.x;
    const dy = s.y - poi.y;
    return {
      index,
      shot: s,
      distPx: Math.sqrt(dx * dx + dy * dy)
    };
  });

  // 3. Compute Mean and Standard Deviation of Radii
  const meanR = radii.reduce((sum, r) => sum + r.distPx, 0) / n;
  const variance = radii.reduce((sum, r) => sum + Math.pow(r.distPx - meanR, 2), 0) / (n - 1);
  const stdDevR = Math.sqrt(variance);

  if (stdDevR === 0) return { cleanShots: shots, flyers: [] };

  // 4. Find max distance shot
  let maxDiff = -1;
  let candidate = null;

  radii.forEach((r) => {
    const diff = Math.abs(r.distPx - meanR);
    if (diff > maxDiff) {
      maxDiff = diff;
      candidate = r;
    }
  });

  // 5. Compute G statistic
  const G = maxDiff / stdDevR;

  // Critical values table for Grubbs' Test (alpha = 0.05)
  const gCriticalTable = {
    3: 1.153, 4: 1.463, 5: 1.672, 6: 1.822, 7: 1.938,
    8: 2.032, 9: 2.110, 10: 2.176, 12: 2.285, 15: 2.409
  };

  const gCrit = gCriticalTable[Math.min(n, 15)] || 2.2;

  // 6. Evaluate outlier
  if (G > gCrit) {
    const flyers = [candidate.shot];
    const cleanShots = shots.filter((_, idx) => idx !== candidate.index);
    return { cleanShots, flyers, outlierIndex: candidate.index, gScore: G.toFixed(2) };
  }

  return { cleanShots: shots, flyers: [] };
};

    .onEnd(() => {
      savedTranslateX.value = translateX.value;
      savedTranslateY.value = translateY.value;
    });

  // 3. Single-Finger Precision Tap / Drag Loupe
  const tapGesture = Gesture.Tap()
    .maxDuration(250)
    .onTouchesDown((e) => {
      if (e.allTouches.length === 1) {
        touchX.value = e.allTouches[0].x;
        touchY.value = e.allTouches[0].y;
        isTouching.value = true;
      }
    })
    .onTouchesUp((e) => {
      isTouching.value = false;
      const x = (e.allTouches[0].x - translateX.value) / scale.value;
      const y = (e.allTouches[0].y - translateY.value) / scale.value;
      
      runOnJS(registerShot)({ x, y });
    });

  const registerShot = (pt) => {
    const updated = [...shots, pt];
    setShots(updated);
    if (onAddShot) onAddShot(updated);
  };

  const composedGesture = Gesture.Race(
    Gesture.Simultaneous(pinchGesture, panGesture),
    tapGesture
  );

  // Animated Styles
  const canvasStyle = useAnimatedStyle(() => ({
    transform: [
      { translateX: translateX.value },
      { translateY: translateY.value },
      { scale: scale.value },
    ],
  }));

  const loupeStyle = useAnimatedStyle(() => {
    return {
      opacity: isTouching.value ? 1 : 0,
      transform: [
        { translateX: touchX.value - LOUPE_SIZE / 2 },
        // Offset 80px above finger so touch point isn't obscured
        { translateY: touchY.value - LOUPE_SIZE / 2 - 80 }, 
      ],
    };
  });

  const magnifiedImageStyle = useAnimatedStyle(() => ({
    width: SCREEN_WIDTH * scale.value * LOUPE_MAGNIFICATION,
    height: CANVAS_HEIGHT * scale.value * LOUPE_MAGNIFICATION,
    transform: [
      {
        translateX:
          -touchX.value * LOUPE_MAGNIFICATION + LOUPE_SIZE / 2,
      },
      {
        translateY:
          -touchY.value * LOUPE_MAGNIFICATION + LOUPE_SIZE / 2,
      },
    ],
  }));

  return (
    <GestureHandlerRootView style={styles.container}>
      <GestureDetector gesture={composedGesture}>
        <View style={styles.canvasContainer}>
          {/* Main Image View */}
          <Animated.View style={[styles.imageWrapper, canvasStyle]}>
            <Image source={{ uri: imageUri }} style={styles.image} resizeMode="contain" />
            
            {/* Render Registered Shots */}
            {shots.map((s, idx) => (
              <View
                key={idx}
                style={[styles.shotMarker, { left: s.x - 8, top: s.y - 8 }]}
              >
                <Text style={styles.shotText}>{idx + 1}</Text>
              </View>
            ))}
          </Animated.View>

          {/* Floating Loupe Lens */}
          <Animated.View style={[styles.loupe, loupeStyle]} pointerEvents="none">
            <Animated.Image

            <View style={styles.loupeCrosshairV} />
          </Animated.View>
        </View>
      </GestureDetector>
    </GestureHandlerRootView>
  );

// 1. Calculate Circular Error Probable (CEP)
export const calculateCEP = (shots, poi, ppi) => {
  if (shots.length < 2 || !ppi) return { cep50: 0, cep95: 0 };

  const sigmaX = Math.sqrt(
    shots.reduce((acc, s) => acc + Math.pow((s.x - poi.x) / ppi, 2), 0) / (shots.length - 1)
  );
  const sigmaY = Math.sqrt(
    shots.reduce((acc, s) => acc + Math.pow((s.y - poi.y) / ppi, 2), 0) / (shots.length - 1)
  );

  // Rayleigh distribution approximations
  const cep50 = 0.5887 * (sigmaX + sigmaY);
  const cep95 = 1.241 * (sigmaX + sigmaY);

  return { cep50: cep50.toFixed(2), cep95: cep95.toFixed(2) };
};

// 2. Grubbs' Test for Outlier/Flyer Detection
export const filterFlyersGrubbs = (shots, ppi) => {
  if (shots.length < 3) return { cleanShots: shots, flyer: null };

  const n = shots.length;
  const poi = shots.reduce(
    (acc, s) => ({ x: acc.x + s.x / n, y: acc.y + s.y / n }),
    { x: 0, y: 0 }
  );

  const radii = shots.map((s, idx) => ({
    idx,
    shot: s,
    r: Math.sqrt(Math.pow(s.x - poi.x, 2) + Math.pow(s.y - poi.y, 2)) / ppi,
  }));

  const meanR = radii.reduce((a, b) => a + b.r, 0) / n;
  const stdR = Math.sqrt(
    radii.reduce((a, b) => a + Math.pow(b.r - meanR, 2), 0) / (n - 1)
  );

  let maxG = 0;
  let candidate = null;

  radii.forEach((item) => {
    const G = Math.abs(item.r - meanR) / stdR;
    if (G > maxG) {
      maxG = G;
      candidate = item;
    }
  });

  // Critical values for alpha = 0.05
  const gCritMap = { 3: 1.15, 4: 1.46, 5: 1.67, 6: 1.82, 7: 1.94, 8: 2.03, 9: 2.11, 10: 2.18 };
  const gCrit = gCritMap[Math.min(n, 10)] || 2.2;

  if (maxG > gCrit) {
    return {
      cleanShots: shots.filter((_, i) => i !== candidate.idx),
      flyer: candidate.shot,
      gScore: maxG.toFixed(2),
    };
  }

  return { cleanShots: shots, flyer: null };
};

// 3. Multi-Group Load Comparison
export const compareLoads = (groups) => {
  // groups = [{ loadName: 'Load A', shots: [...], ppi: 100 }, ...]
  return groups.map((g) => {
    const { cleanShots, flyer } = filterFlyersGrubbs(g.shots, g.ppi);
    const poi = cleanShots.reduce(
      (acc, s) => ({ x: acc.x + s.x / cleanShots.length, y: acc.y + s.y / cleanShots.length }),
      { x: 0, y: 0 }
    );
    const cep = calculateCEP(cleanShots, poi, g.ppi);

    return {
      loadName: g.loadName,
      totalShots: g.shots.length,
      hasFlyer: !!flyer,
      cep50: cep.cep50,
      cep95: cep.cep95,
    };
  });
};

/**
 * Decomposes group dispersion into Vertical (Velocity/Harmonics) vs Horizontal (Wind/Trigger)
 */
export const analyzeDispersionVectors = (shots, ppi, velocitySD = null) => {
  if (shots.length < 3) return null;

  const n = shots.length;
  const meanX = shots.reduce((sum, s) => sum + s.x, 0) / n;
  const meanY = shots.reduce((sum, s) => sum + s.y, 0) / n;

  // Calculate Variance in Inches
  const varX = shots.reduce((sum, s) => sum + Math.pow((s.x - meanX) / ppi, 2), 0) / (n - 1);
  const varY = shots.reduce((sum, s) => sum + Math.pow((s.y - meanY) / ppi, 2), 0) / (n - 1);

  const stdDevX = Math.sqrt(varX); // Horizontal spread
  const stdDevY = Math.sqrt(varY); // Vertical spread

  const aspectRatio = stdDevY / (stdDevX || 0.001); // Vertical / Horizontal ratio

  let primaryDiagnosis = 'Balanced Dispersion (Good Mechanical & Load Balance)';
  if (aspectRatio > 1.6) {
    primaryDiagnosis = velocitySD && velocitySD > 15 
      ? `High Vertical Dispersion driven by Muzzle Velocity SD (${velocitySD} fps). Work on neck tension or powder charge consistency.`
      : 'High Vertical Dispersion. Check rifle barrel heating, stock forend pressure, or breathing control.';
  } else if (aspectRatio < 0.6) {
    primaryDiagnosis = 'High Horizontal Dispersion. Check wind read consistency, scope level (cant), or lateral trigger torque.';
  }

  return {
    horizontalStdDevInches: stdDevX.toFixed(2),
    verticalStdDevInches: stdDevY.toFixed(2),
    aspectRatio: aspectRatio.toFixed(2),
    primaryDiagnosis,
  };
};

/**
 * Calculates Scope Tracking Error Percentage from Tall Target Test
 */
export const calculateScopeTrackingError = ({
  bottomPoiYPx,
  topPoiYPx,
  ppi,
  dialedUnits, // e.g. 30 MOA or 10 MIL
  unit = 'MOA',
  distanceYards = 100
}) => {
  const measuredHeightInches = Math.abs(topPoiYPx - bottomPoiYPx) / ppi;
  const angularFactor = unit === 'MOA' ? 0.010472 : 0.036;
  const expectedHeightInches = distanceYards * angularFactor * dialedUnits;

  const trackingCorrectionMultiplier = measuredHeightInches / expectedHeightInches;
  const errorPercentage = (trackingCorrectionMultiplier - 1) * 100;

  return {
    measuredHeightInches: measuredHeightInches.toFixed(2),
    expectedHeightInches: expectedHeightInches.toFixed(2),
    multiplier: trackingCorrectionMultiplier.toFixed(4), // e.g., 0.9750
    errorPercentage: `${errorPercentage > 0 ? '+' : ''}${errorPercentage.toFixed(1)}%`,
    recommendation: Math.abs(errorPercentage) > 1.5
      ? `Scope is under/over-tracking. Apply a scale factor of ${trackingCorrectionMultiplier.toFixed(3)} to ballistic solver output.`
      : 'Scope tracking is within acceptable 1.5% tolerance.'
  };

/**
 * Calculates Powder Temperature Sensitivity Coefficient from two chronograph logs

export const calculatePowderTempSensitivity = (logCold, logHot) => {
  // logCold: { tempF: 35, velocityFps: 2680 }
  // logHot:  { tempF: 85, velocityFps: 2730 }
  
  const deltaTemp = logHot.tempF - logCold.tempF;
  const deltaVel = logHot.velocityFps - logCold.velocityFps;

  if (deltaTemp <= 0) return null;

  const coefficient = deltaVel / deltaTemp; // fps per degree F

  let rating = 'Extremely Temperature Stable';
  if (coefficient > 1.2) rating = 'Temperature Sensitive (Significant winter/summer drop shift)';
  else if (coefficient > 0.6) rating = 'Moderate Sensitivity';

  return {
    fpsPerDegreeF: coefficient.toFixed(2),
    rating,
    predictedVelAtZeroF: (logCold.velocityFps - (logCold.tempF * coefficient)).toFixed(0),
    predictedVelAt100F: (logCold.velocityFps + ((100 - logCold.tempF) * coefficient)).toFixed(0),
  };
};

/**
 * Warns if the bullet crosses the transonic boundary before reaching target distance

export const evaluateTransonicStability = (velocityFps, tempF, distanceYards, Sg) => {
  const speedOfSoundFps = 49.022 * Math.sqrt(tempF + 459.67);
  const transonicCutoffFps = 1.2 * speedOfSoundFps;

  // Simple velocity decay estimate over distance
  const estimatedImpactVelocity = velocityFps - (distanceYards * 0.85);

  const isTransonic = estimatedImpactVelocity <= transonicCutoffFps;
  const isUnstableInTransonic = isTransonic && Sg < 1.5;

  return {
    speedOfSoundFps: speedOfSoundFps.toFixed(0),
    transonicCutoffFps: transonicCutoffFps.toFixed(0),
    estimatedImpactVelocity: Math.max(0, estimatedImpactVelocity).toFixed(0),
    isTransonic,
    statusMessage: isUnstableInTransonic
      ? `WARNING: Bullet enters transonic region at ~${distanceYards}yd with Sg=${Sg}. Expect severe stability loss and vertical stringing.`
      : isTransonic
      ? `INFO: Bullet is transonic at target (${estimatedImpactVelocity} fps). High Sg (${Sg}) should maintain stability.`
      : 'STABLE: Bullet remains supersonic through target distance.'
  };

/**
 * Calculates Rayleigh Statistical Metrics for a shot string
 */
export const calculateRayleighGroupMetrics = (shotsInInches) => {
  const n = shotsInInches.length;
  if (n < 3) return null;

  const meanX = shotsInInches.reduce((sum, s) => sum + s.x, 0) / n;
  const meanY = shotsInInches.reduce((sum, s) => sum + s.y, 0) / n;

  // Calculate radius of each shot from center
  const radii = shotsInInches.map(s => Math.sqrt(Math.pow(s.x - meanX, 2) + Math.pow(s.y - meanY, 2)));
  const meanRadius = radii.reduce((sum, r) => sum + r, 0) / n;

  return {
    meanRadiusInches: meanRadius.toFixed(2),
    cep50Inches: (1.1774 * meanRadius).toFixed(2), // 50% of shots fall within this radius
    cep90Inches: (2.146 * meanRadius).toFixed(2),  // 90% of shots fall within this radius
    description: `50% of all future shots will land within ${(1.1774 * meanRadius).toFixed(2)}" of target center.`
  };

JavaScript
import React, { useState } from 'react';
import {
  StyleSheet,
  View,
  Text,
  TextInput,
  TouchableOpacity,
  ScrollView,
  Alert,
} from 'react-native';

export default function PowderTempSensitivityScreen() {
  // Session logs: Temp (°F) vs Muzzle Velocity (fps)
  const [logs, setLogs] = useState([
    { id: '1', tempF: '35', velocityFps: '2680' },
    { id: '2', tempF: '85', velocityFps: '2730' },
  ]);

  const [newTemp, setNewTemp] = useState('');
  const [newVel, setNewVel] = useState('');

  // Add new session log
  const addLog = () => {
    if (!newTemp || !newVel) {
      Alert.alert('Missing Input', 'Please enter both temperature and velocity.');
      return;
    }
    setLogs([
      ...logs,
      { id: Date.now().toString(), tempF: newTemp, velocityFps: newVel },
    ]);
    setNewTemp('');
    setNewVel('');
  };

  // Remove a session log
  const removeLog = (id) => {
    if (logs.length <= 2) {
      Alert.alert('Minimum Requirement', 'You need at least 2 sessions to calculate sensitivity.');
      return;
    }
    setLogs(logs.filter((item) => item.id !== id));
  };

  // Perform Least-Squares Linear Regression: V(T) = (C_T * T) + V_0
  const calculateSensitivity = () => {
    const validPoints = logs
      .map((l) => ({ t: parseFloat(l.tempF), v: parseFloat(l.velocityFps) }))
      .filter((p) => !isNaN(p.t) && !isNaN(p.v));

    if (validPoints.length < 2) return null;

    const n = validPoints.length;
    const sumT = validPoints.reduce((acc, p) => acc + p.t, 0);
    const sumV = validPoints.reduce((acc, p) => acc + p.v, 0);
    const sumTV = validPoints.reduce((acc, p) => acc + p.t * p.v, 0);
    const sumT2 = validPoints.reduce((acc, p) => acc + p.t * p.t, 0);

    const denominator = n * sumT2 - sumT * sumT;
    if (denominator === 0) return null; // Avoid division by zero if all temps are identical

    // Slope (C_T) in fps per °F
    const ct = (n * sumTV - sumT * sumV) / denominator;
    
    // Y-intercept (theoretical velocity at 0°F)
    const intercept = (sumV - ct * sumT) / n;

    // Qualitative assessment
    let stabilityRating = 'EXTREMELY STABLE';
    let ratingColor = '#10b981'; // Green
    if (Math.abs(ct) > 1.2) {
      stabilityRating = 'HIGHLY SENSITIVE';
      ratingColor = '#ef4444'; // Red
    } else if (Math.abs(ct) > 0.6) {
      stabilityRating = 'MODERATE SENSITIVITY';
      ratingColor = '#f59e0b'; // Amber
    }

    // Generate predicted velocity matrix (-10°F to 110°F in 20° steps)
    const steps = [-10, 10, 30, 50, 70, 90, 110];
    const predictions = steps.map((temp) => ({
      temp,
      predictedFps: Math.round(intercept + ct * temp),
      deltaFps: Math.round((temp - 70) * ct), // Shift relative to standard 70°F
    }));

import React, { useState } from 'react';
import {
  StyleSheet,
  View,
  Text,
  TouchableOpacity,
  ScrollView,
  Dimensions,
} from 'react-native';

const TARGET_SIZE = 280; // Target Canvas Dimensions (px)
const SCALE_INCHES = 6;  // Total target width in inches (-3" to +3")
const PX_PER_INCH = TARGET_SIZE / SCALE_INCHES;

export default function RayleighGroupCalculator() {
  const [shots, setShots] = useState([
    { id: '1', x: -0.45, y: 0.62 },
    { id: '2', x: 0.35, y: 0.28 },
    { id: '3', x: -0.12, y: -0.55 },
    { id: '4', x: 0.50, y: -0.20 },
    { id: '5', x: -0.28, y: 0.15 },
  ]);

  const [distanceYards, setDistanceYards] = useState(100);

  // Tap handler to add shots directly onto target face
  const handleTargetTap = (event) => {
    const { locationX, locationY } = event.nativeEvent;
    
    // Convert canvas pixels to relative inches centered at (0,0)
    const xInches = (locationX - TARGET_SIZE / 2) / PX_PER_INCH;
    const yInches = (TARGET_SIZE / 2 - locationY) / PX_PER_INCH; // Flip Y for standard Cartesian

    setShots([
      ...shots,
      { id: Date.now().toString(), x: xInches, y: yInches },
    ]);
  };

  const clearShots = () => setShots([]);
  const removeLastShot = () => setShots(shots.slice(0, -1));

  // Compute Rayleigh Group Metrics
  const calculateMetrics = () => {
    const n = shots.length;
    if (n < 3) return null;

    // 1. Center of Impact (COI)
    const coiX = shots.reduce((sum, s) => sum + s.x, 0) / n;
    const coiY = shots.reduce((sum, s) => sum + s.y, 0) / n;

    // 2. Radial distances from COI
    const radii = shots.map((s) =>
      Math.sqrt(Math.pow(s.x - coiX, 2) + Math.pow(s.y - coiY, 2))
    );
    const meanRadius = radii.reduce((sum, r) => sum + r, 0) / n;

    // 3. Extreme Spread (Max distance between any two shots)
    let extremeSpread = 0;
    for (let i = 0; i < n; i++) {
      for (let j = i + 1; j < n; j++) {
        const d = Math.sqrt(
          Math.pow(shots[i].x - shots[j].x, 2) +
          Math.pow(shots[i].y - shots[j].y, 2)
        );
        if (d > extremeSpread) extremeSpread = d;
      }
    }

    // 4. Rayleigh Parameter (Sigma)
    const sumSqRadii = radii.reduce((sum, r) => sum + r * r, 0);
    const sigma = Math.sqrt(sumSqRadii / (2 * n));

    // 5. CEP Radii
    const cep50 = 1.1774 * sigma;
    const cep90 = 2.1460 * sigma;

    // 1 MOA in inches at user distance = 1.047 * (distance / 100)
    const moaInches = 1.047 * (distanceYards / 100);

    return {
      n,
      coiX,
      coiY,
      meanRadius,
      extremeSpread,
      extremeSpreadMOA: extremeSpread / moaInches,
      sigma,
      cep50,
      cep50MOA: cep50 / moaInches,
      cep90,
      cep90MOA: cep90 / moaInches,
    };
  };

  const metrics = calculateMetrics();

  return (
    <ScrollView style={styles.container} contentContainerStyle={styles.content}>
      <Text style={styles.header}>Rayleigh CEP Group Calculator</Text>
      <Text style={styles.sub}>
        Tap target face to log shot impacts. Minimum 3 shots required for statistical estimation.
      </Text>

      {/* Interactive Target Canvas */}
      <View style={styles.targetWrapper}>
        <TouchableOpacity
          activeOpacity={0.9}
          style={styles.targetCanvas}
          onPress={handleTargetTap}
        >
          {/* Target Grid Rings */}
          <View style={[styles.ring, { width: PX_PER_INCH * 4, height: PX_PER_INCH * 4 }]} />
          <View style={[styles.ring, { width: PX_PER_INCH * 2, height: PX_PER_INCH * 2 }]} />
          <View style={styles.crosshairH} />
          <View style={styles.crosshairV} />

          {/* Render CEP 50% Overlay Ring if metrics available */}
          {metrics && (
            <View
              style={[
                styles.cepRing,
                {
                  width: metrics.cep50 * 2 * PX_PER_INCH,
                  height: metrics.cep50 * 2 * PX_PER_INCH,
                  borderRadius: metrics.cep50 * PX_PER_INCH,
                  left: TARGET_SIZE / 2 + metrics.coiX * PX_PER_INCH - metrics.cep50 * PX_PER_INCH,
                  top: TARGET_SIZE / 2 - metrics.coiY * PX_PER_INCH - metrics.cep50 * PX_PER_INCH,
                },
              ]}
            />
          )}

          {/* Render Shot Dots */}
          {shots.map((shot, idx) => {
            const pxX = TARGET_SIZE / 2 + shot.x * PX_PER_INCH - 6;
            const pxY = TARGET_SIZE / 2 - shot.y * PX_PER_INCH - 6;
            return (
              <View key={shot.id} style={[styles.shotDot, { left: pxX, top: pxY }]}>
                <Text style={styles.shotNumber}>{idx + 1}</Text>
              </View>
            );
          })}

          {/* Render Center of Impact (COI) Marker */}
          {metrics && (
            <View
              style={[
                styles.coiMarker,
                {
                  left: TARGET_SIZE / 2 + metrics.coiX * PX_PER_INCH - 5,
                  top: TARGET_SIZE / 2 - metrics.coiY * PX_PER_INCH - 5,
                },
              ]}
            />
          )}
        </TouchableOpacity>

        <View style={styles.controlsRow}>
          <TouchableOpacity style={styles.secondaryBtn} onPress={removeLastShot}>
            <Text style={styles.secondaryBtnText}>Undo Shot</Text>
          </TouchableOpacity>
          <TouchableOpacity style={styles.dangerBtn} onPress={clearShots}>
            <Text style={styles.dangerBtnText}>Reset Target</Text>
          </TouchableOpacity>
        </View>
      </View>

      {/* Statistical Results Engine */}
      {metrics ? (
        <View style={styles.card}>
          <Text style={styles.cardTitle}>RAYLEIGH GROUP METRICS ({metrics.n} SHOTS)</Text>

          <View style={styles.metricGrid}>

            <View style={styles.gridItem}>

              </Text>
              <Text style={styles.metricSub}>
                {metrics.cep90MOA.toFixed(2)} MOA
              </Text>

            {/* Mean Radius */}
            <View style={styles.gridItem}>
              <Text style={styles.metricLabel}>MEAN RADIUS (MR)</Text>
              <Text style={styles.metricValue}>
                {metrics.meanRadius.toFixed(2)}"
              </Text>
            </View>

            {/* Traditional Extreme Spread */}
            <View style={styles.gridItem}>
              <Text style={styles.metricLabel}>EXTREME SPREAD (ES)</Text>
              <Text style={styles.metricValue}>

To implement offline persistence for rifles, chronograph logs, and shot groups, you can use SQLite. The example below uses expo-sqlite (which works in both Expo and bare React Native apps) with standard relational foreign keys.

2. Database Layer: services/database.js
This service creates the database tables, enables foreign key constraints, and provides async methods to read and write records.

JavaScript
import * as SQLite from 'expo-sqlite';

let db;

// Initialize Database and Create Tables
export const initDatabase = async () => {
  db = await SQLite.openDatabaseAsync('ballistics.db');

  // Enable foreign key constraints
  await db.execAsync('PRAGMA foreign_keys = ON;');

  // Initialize schema
  await db.execAsync(`
    CREATE TABLE IF NOT EXISTS rifles (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      name TEXT NOT NULL,
      caliber TEXT NOT NULL,
      zero_distance_yd REAL DEFAULT 100,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP
    );

    CREATE TABLE IF NOT EXISTS chronograph_logs (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      rifle_id INTEGER NOT NULL,
      temp_f REAL NOT NULL,
      velocity_fps REAL NOT NULL,
      powder_name TEXT,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (rifle_id) REFERENCES rifles (id) ON DELETE CASCADE
    );

    CREATE TABLE IF NOT EXISTS shot_groups (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      rifle_id INTEGER NOT NULL,
      distance_yd REAL NOT NULL,
      cep_50_inches REAL NOT NULL,
      extreme_spread_inches REAL NOT NULL,
      shot_count INTEGER NOT NULL,
      created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (rifle_id) REFERENCES rifles (id) ON DELETE CASCADE
    );
  `);
};

// --- RIFLE QUERIES ---
export const getRifles = async () => {
  return await db.getAllAsync('SELECT * FROM rifles ORDER BY id DESC;');
};

export const addRifle = async (name, caliber, zeroDistanceYd = 100) => {
  const result = await db.runAsync(
    'INSERT INTO rifles (name, caliber, zero_distance_yd) VALUES (?, ?, ?);',
    [name, caliber, zeroDistanceYd]
  );
  return result.lastInsertRowId;
};

// --- CHRONOGRAPH QUERIES ---
export const addChronoLog = async (rifleId, tempF, velocityFps, powderName = '') => {
  const result = await db.runAsync(
    'INSERT INTO chronograph_logs (rifle_id, temp_f, velocity_fps, powder_name) VALUES (?, ?, ?, ?);',
    [rifleId, tempF, velocityFps, powderName]
  );
  return result.lastInsertRowId;
};

export const getChronoLogsByRifle = async (rifleId) => {
  return await db.getAllAsync(
    'SELECT * FROM chronograph_logs WHERE rifle_id = ? ORDER BY created_at DESC;',
    [rifleId]
  );
};

// --- SHOT GROUP QUERIES ---
export const addShotGroup = async (rifleId, distanceYd, cep50, extremeSpread, shotCount) => {
  const result = await db.runAsync(
    `INSERT INTO shot_groups (rifle_id, distance_yd, cep_50_inches, extreme_spread_inches, shot_count) 
     VALUES (?, ?, ?, ?, ?);`,
    [rifleId, distanceYd, cep50, extremeSpread, shotCount]
  );
  return result.lastInsertRowId;
};

export const getShotGroupsByRifle = async (rifleId) => {
  return await db.getAllAsync(
    'SELECT * FROM shot_groups WHERE rifle_id = ? ORDER BY created_at DESC;',
    [rifleId]
  );
};
3. Custom React Hook: hooks/useBallisticsDb.js
This hook handles database loading states and re-fetches records whenever a new rifle, log, or shot group is saved.

JavaScript
import { useState, useEffect, useCallback } from 'react';
import {
  initDatabase,
  getRifles,
  addRifle,
  addChronoLog,
  getChronoLogsByRifle,
  addShotGroup,
  getShotGroupsByRifle,
} from '../services/database';

export function useBallisticsDb() {
  const [isReady, setIsReady] = useState(false);
  const [rifles, setRifles] = useState([]);
  const [selectedRifleId, setSelectedRifleId] = useState(null);
  const [chronoLogs, setChronoLogs] = useState([]);
  const [shotGroups, setShotGroups] = useState([]);

  // Init DB on startup
  useEffect(() => {
    async function setup() {
      await initDatabase();
      const loadedRifles = await getRifles();
      setRifles(loadedRifles);
      if (loadedRifles.length > 0) {
        setSelectedRifleId(loadedRifles[0].id);
      }
      setIsReady(true);
    }
    setup();
  }, []);

  // Fetch relational data when selected rifle changes
  const refreshRifleData = useCallback(async (rifleId) => {
    if (!rifleId) return;
    const [logs, groups] = await Promise.all([
      getChronoLogsByRifle(rifleId),
      getShotGroupsByRifle(rifleId),
    ]);
    setChronoLogs(logs);
    setShotGroups(groups);
  }, []);

  useEffect(() => {
    if (selectedRifleId) {
      refreshRifleData(selectedRifleId);
    }
  }, [selectedRifleId, refreshRifleData]);

  // Actions
  const handleCreateRifle = async (name, caliber, zeroDistance) => {
    const newId = await addRifle(name, caliber, zeroDistance);
    const updatedRifles = await getRifles();
    setRifles(updatedRifles);
    setSelectedRifleId(newId);
  };

  const handleLogChrono = async (tempF, velocityFps, powderName) => {
    if (!selectedRifleId) return;
    await addChronoLog(selectedRifleId, tempF, velocityFps, powderName);
    await refreshRifleData(selectedRifleId);
  };

  const handleSaveGroup = async (distanceYd, cep50, extremeSpread, shotCount) => {
    if (!selectedRifleId) return;
    await addShotGroup(selectedRifleId, distanceYd, cep50, extremeSpread, shotCount);
    await refreshRifleData(selectedRifleId);
  };

  return {
    isReady,
    rifles,
    selectedRifleId,
    setSelectedRifleId,
    chronoLogs,
    shotGroups,
    createRifle: handleCreateRifle,
    logChrono: handleLogChrono,
    saveGroup: handleSaveGroup,
  };
}

import * as FileSystem from 'expo-file-system';
import * as Sharing from 'expo-sharing';
import * as DocumentPicker from 'expo-document-picker';
import * as SQLite from 'expo-sqlite';

// --- Helper to obtain DB instance ---
const getDb = async () => await SQLite.openDatabaseAsync('ballistics.db');

// ==========================================
// 1. JSON FULL DATABASE BACKUP & RESTORE
// ==========================================

export const exportDatabaseToJson = async () => {
  try {
    const db = await getDb();

    // 1. Query all tables
    const rifles = await db.getAllAsync('SELECT * FROM rifles;');
    const chronoLogs = await db.getAllAsync('SELECT * FROM chronograph_logs;');
    const shotGroups = await db.getAllAsync('SELECT * FROM shot_groups;');

    // 2. Build structured JSON payload
    const backupData = {
      version: 1,
      exportedAt: new Date().toISOString(),
      data: {
        rifles,
        chronograph_logs: chronoLogs,
        shot_groups: shotGroups,
      },
    };

    const jsonString = JSON.stringify(backupData, null, 2);

    // 3. Write file to local app directory
    const fileUri = `${FileSystem.documentDirectory}ballistics_backup_${Date.now()}.json`;
    await FileSystem.writeAsStringAsync(fileUri, jsonString, {
      encoding: FileSystem.EncodingType.UTF8,
    });

    // 4. Open native Share sheet
    if (await Sharing.isAvailableAsync()) {
      await Sharing.shareAsync(fileUri, {
        mimeType: 'application/json',
        dialogTitle: 'Export Ballistics JSON Backup',
      });
    }

    return { success: true, fileUri };
  } catch (error) {
    console.error('JSON Export Error:', error);
    return { success: false, error: error.message };
  }
};

export const importDatabaseFromJson = async () => {
  try {
    // 1. Pick file
    const pickerResult = await DocumentPicker.getDocumentAsync({
      type: 'application/json',
      copyToCacheDirectory: true,
    });

    if (pickerResult.canceled) return { success: false, message: 'Cancelled' };

    const selectedFile = pickerResult.assets[0];

    // 2. Read contents
    const jsonString = await FileSystem.readAsStringAsync(selectedFile.uri, {
      encoding: FileSystem.EncodingType.UTF8,
    });

    const backupData = JSON.parse(jsonString);

    if (!backupData.data || !backupData.data.rifles) {
      throw new Error('Invalid backup file format.');
    }

    const db = await getDb();

    // 3. Import data within an atomic transaction
    await db.withTransactionAsync(async () => {
      const { rifles, chronograph_logs, shot_groups } = backupData.data;

      // Insert Rifles
      for (const r of rifles || []) {
        await db.runAsync(
          'INSERT OR REPLACE INTO rifles (id, name, caliber, zero_distance_yd, created_at) VALUES (?, ?, ?, ?, ?);',
          [r.id, r.name, r.caliber, r.zero_distance_yd, r.created_at]
        );
      }

      // Insert Chronograph Logs
      for (const c of chronograph_logs || []) {
        await db.runAsync(
          'INSERT OR REPLACE INTO chronograph_logs (id, rifle_id, temp_f, velocity_fps, powder_name, created_at) VALUES (?, ?, ?, ?, ?, ?);',
          [c.id, c.rifle_id, c.temp_f, c.velocity_fps, c.powder_name, c.created_at]
        );
      }

      // Insert Shot Groups
      for (const g of shot_groups || []) {
        await db.runAsync(
          'INSERT OR REPLACE INTO shot_groups (id, rifle_id, distance_yd, cep_50_inches, extreme_spread_inches, shot_count, created_at) VALUES (?, ?, ?, ?, ?, ?, ?);',
          [g.id, g.rifle_id, g.distance_yd, g.cep_50_inches, g.extreme_spread_inches, g.shot_count, g.created_at]
        );
      }
    });

    return { success: true };
  } catch (error) {
    console.error('JSON Import Error:', error);
    return { success: false, error: error.message };
  }
};

export const exportChronoLogsToCsv = async () => {
  try {
    const db = await getDb();
    
    // Join with Rifle name for human-readable CSV export
    const rows = await db.getAllAsync(`
      SELECT c.id, r.name AS rifle_name, c.temp_f, c.velocity_fps, c.powder_name, c.created_at 
      FROM chronograph_logs c 
      JOIN rifles r ON c.rifle_id = r.id 
      ORDER BY c.created_at DESC;
    `);

    // Build CSV Content
    const headers = ['ID', 'Rifle Name', 'Temp (F)', 'Velocity (FPS)', 'Powder Name', 'Logged At'];
    const csvLines = [headers.join(',')];

    rows.forEach((row) => {
      const line = [
        row.id,
        `"${row.rifle_name.replace(/"/g, '""')}"`,
        row.temp_f,
        row.velocity_fps,
        `"${(row.powder_name || '').replace(/"/g, '""')}"`,
        `"${row.created_at}"`,
      ].join(',');
      csvLines.push(line);
    });

    const csvString = csvLines.join('\n');
    const fileUri = `${FileSystem.documentDirectory}chronograph_logs_${Date.now()}.csv`;

    await FileSystem.writeAsStringAsync(fileUri, csvString, {
      encoding: FileSystem.EncodingType.UTF8,
    });

    if (await Sharing.isAvailableAsync()) {
      await Sharing.shareAsync(fileUri, {
        mimeType: 'text/csv',
        dialogTitle: 'Export Chronograph CSV',
      });
    }

    return { success: true };
  } catch (error) {
    console.error('CSV Export Error:', error);
    return { success: false, error: error.message };
  }
};

export const importChronoLogsFromCsv = async (targetRifleId) => {
  try {
    const pickerResult = await DocumentPicker.getDocumentAsync({
      type: ['text/csv', 'text/comma-separated-values', 'text/plain'],
      copyToCacheDirectory: true,
    });

    if (pickerResult.canceled) return { success: false, message: 'Cancelled' };

    const fileUri = pickerResult.assets[0].uri;
    const csvString = await FileSystem.readAsStringAsync(fileUri, {
      encoding: FileSystem.EncodingType.UTF8,
    });

    const lines = csvString.split(/\r?\n/).filter((line) => line.trim() !== '');
    if (lines.length < 2) throw new Error('CSV file is empty or missing data rows.');

    const db = await getDb();

    // Skip Header row (index 0) and parse rows
    await db.withTransactionAsync(async () => {
      for (let i = 1; i < lines.length; i++) {
        // Rudimentary CSV parse splitting on comma (ignoring quotes for standard log lines)
        const cols = lines[i].split(',').map((col) => col.replace(/^"|"$/g, '').trim());
        
        // Expected format: [ID, RifleName, TempF, VelocityFps, PowderName, Date]
        if (cols.length >= 4) {
          const tempF = parseFloat(cols[2]) || 70;
          const velocityFps = parseFloat(cols[3]) || 0;
          const powderName = cols[4] || 'CSV Import';

          if (velocityFps > 0) {
            await db.runAsync(
              'INSERT INTO chronograph_logs (rifle_id, temp_f, velocity_fps, powder_name) VALUES (?, ?, ?, ?);',
              [targetRifleId, tempF, velocityFps, powderName]
            );
          }
        }
      }
    });

    return { success: true };
  } catch (error) {
    console.error('CSV Import Error:', error);
    return { success: false, error: error.message };
  }
};

Build-Time Tooling: The vulnerable uuid package is a nested dependency of xcode, which is used internally by @expo/config-plugins and @expo/cli. It is only executed on your local machine or CI server during native builds—it is not bundled into your client-side production app code.

Npm Dependency Resolution: npm flags nested sub-dependencies even when parent CLI tools haven't updated their package.json dependency ranges yet.

Solution 1: Use overrides in package.json (Recommended)
You can force npm to resolve all sub-dependencies of uuid to a patched version without changing your top-level Expo SDK packages.

Open your package.json file.

Add an overrides field at the root level (or resolutions if using Yarn):

JSON
{
  "name": "your-ballistics-app",
  "version": "1.0.0",
  "main": "expo/AppEntry.js",
  "scripts": { ... },
  "dependencies": { ... },
  "devDependencies": { ... },
  "overrides": {
    "uuid": "^11.1.1"
  }
}
